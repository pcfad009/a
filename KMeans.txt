# K-Means clustering + elbow method. Any NUMERIC dataset works.
from pathlib import Path
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

FILE = 'data.csv'   # CHANGE to examiner's CSV; otherwise built-in Iris
if Path(FILE).exists():
    df = pd.read_csv(FILE)
    # Exclude obvious identifiers, because they are not meaningful features.
    df = df.drop(columns=['Id', 'ID', 'Order ID', 'ORDERNUMBER'], errors='ignore')
    X = df.select_dtypes(include='number').dropna().values
else:
    X = load_iris().data
X = StandardScaler().fit_transform(X)
max_k = min(10, len(X))
inertia = []
for k in range(1, max_k + 1):
    model = KMeans(n_clusters=k, random_state=42, n_init=10)
    model.fit(X)
    inertia.append(model.inertia_)
plt.plot(range(1, max_k + 1), inertia, marker='o')
plt.xlabel('Number of clusters (k)')
plt.ylabel('Inertia'); plt.title('Elbow Method'); plt.show()
k = 3  # CHANGE after reading the elbow graph (example: 3)
k = min(k, len(X))
labels = KMeans(n_clusters=k, random_state=42, n_init=10).fit_predict(X)
plt.scatter(X[:, 0], X[:, 1], c=labels)
plt.xlabel('Feature 1'); plt.ylabel('Feature 2')
plt.title('K-Means Clusters'); plt.show()
print('First 10 group labels:', labels[:10])
