grid = [
 [5,3,0,0,7,0,0,0,0], [6,0,0,1,9,5,0,0,0],
 [0,9,8,0,0,0,0,6,0], [8,0,0,0,6,0,0,0,3],
 [4,0,0,8,0,3,0,0,1], [7,0,0,0,2,0,0,0,6],
 [0,6,0,0,0,0,2,8,0], [0,0,0,4,1,9,0,0,5],
 [0,0,0,0,8,0,0,7,9]
]

def valid(r, c, num):
    if num in grid[r] or any(grid[i][c] == num for i in range(9)):
        return False
    a, b = r//3*3, c//3*3
    return all(grid[i][j] != num for i in range(a, a+3)
               for j in range(b, b+3))

def solve():
    for r in range(9):
        for c in range(9):
            if grid[r][c] == 0:
                for num in range(1, 10):
                    if valid(r, c, num):
                        grid[r][c] = num       # try
                        if solve(): return True
                        grid[r][c] = 0         # undo
                return False
    return True

if solve():
    for row in grid: print(*row)
