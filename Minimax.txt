# Computer is O, human is X. Choose squares 1 to 9.
board = [' '] * 9
wins = [(0,1,2), (3,4,5), (6,7,8), (0,3,6),
        (1,4,7), (2,5,8), (0,4,8), (2,4,6)]

def result():
    for a, b, c in wins:
        if board[a] != ' ' and board[a] == board[b] == board[c]:
            return 1 if board[a] == 'O' else -1
    return 0 if ' ' not in board else None

def minimax(player):
    ans = result()
    if ans is not None:
        return ans
    scores = []
    for i in range(9):
        if board[i] == ' ':
            board[i] = player                 # TRY
            scores.append(minimax('X' if player == 'O' else 'O'))
            board[i] = ' '                    # UNDO
    return max(scores) if player == 'O' else min(scores)

while result() is None:
    print(board[:3], board[3:6], board[6:], sep='\n')
    try:
        pos = int(input('Your square (1-9): ')) - 1
    except ValueError:
        continue
    if not 0 <= pos < 9 or board[pos] != ' ':
        print('Invalid square'); continue
    board[pos] = 'X'
    if result() is not None:
        break
    best = -2
    move = None
    for i in range(9):
        if board[i] == ' ':
            board[i] = 'O'
            score = minimax('X')
            board[i] = ' '
            if score > best:
                best, move = score, i
    board[move] = 'O'
print(board[:3], board[3:6], board[6:], sep='\n')
print({1:'Computer wins', -1:'You win', 0:'Draw'}[result()])
